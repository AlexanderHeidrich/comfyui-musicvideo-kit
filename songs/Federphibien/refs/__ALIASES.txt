# Which words in a shot description mean a reference is IN that shot.
# The files are named in German, the shot descriptions are written in English,
# so without this every character reads as absent. `<slug>: word, word`.
# A `-phrase` is blanked out of the text first, so "hen" does not match inside
# "hen house". A scene no keyword can get right names its own "cast" in
# content.json instead.

das huhn:      hen, hen's, -hen house
der forscher:  researcher, researcher's, scientist, straw hat, folding chair
